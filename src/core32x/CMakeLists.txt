cmake_minimum_required(VERSION 3.22)
project(Kega32xComponents LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
# Standalone component checks; the application loads these devices through GPX.
add_library(kega32x_bus STATIC bus.c)
target_include_directories(kega32x_bus PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_library(kega32x_video STATIC video/compositor.c)
target_include_directories(kega32x_video PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
add_library(kega32x_ntsc565 STATIC video/ntsc565.c)
target_include_directories(kega32x_ntsc565 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..)
if(NOT MSVC)
    target_link_libraries(kega32x_ntsc565 PUBLIC m)
endif()
add_library(kega32x_sh2 STATIC sh2/sh2_interpreter.c sh2/soc.c)
target_include_directories(kega32x_sh2 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/..
    ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party
    ${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/picodrive32x)
add_library(kega32x_pwm STATIC audio/pwm.c)
target_include_directories(kega32x_pwm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/..)
option(KEGA32X_ENABLE_DRC "Enable the experimental x86-64 SH2 DRC backend" OFF)
if(KEGA32X_ENABLE_DRC)
    if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR
       NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
        message(FATAL_ERROR "The SH2 DRC currently supports x86-64 GCC / MinGW only")
    endif()
    target_sources(kega32x_sh2 PRIVATE sh2/drc_backend.c sh2/drc_memory.c)
    target_include_directories(kega32x_sh2 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/sh2)
    target_compile_definitions(kega32x_sh2 PUBLIC KEGA_SH2_DRC=1 PRIVATE DRC_SH2=1)
    target_compile_options(kega32x_sh2 PRIVATE -fno-lto -fno-strict-aliasing -fwrapv)
endif()
add_library(kega32x_machine STATIC machine.c state.c)
target_link_libraries(kega32x_machine PUBLIC kega32x_bus kega32x_sh2 kega32x_pwm)
option(KEGA32X_BUILD_PROBES "Build read-only development device probe" OFF)
if(KEGA32X_BUILD_PROBES)
    add_executable(core32x_device_probe ../../tools/32x-device-probe.c)
    target_link_libraries(core32x_device_probe PRIVATE kega32x_machine)
endif()
include(CTest)
if(BUILD_TESTING)
    if(KEGA32X_ENABLE_DRC)
        add_executable(sh2_drc_tests ../../tests/sh2_drc_tests.c)
        target_link_libraries(sh2_drc_tests PRIVATE kega32x_sh2)
        add_test(NAME core32x_drc COMMAND sh2_drc_tests)
    endif()
    add_executable(kega32x_state_tests ../../tests/core32x_state_tests.c)
    target_link_libraries(kega32x_state_tests PRIVATE kega32x_machine)
    add_test(NAME core32x_state COMMAND kega32x_state_tests)
    add_executable(core32x_bus_tests ../../tests/core32x_bus_tests.c)
    target_link_libraries(core32x_bus_tests PRIVATE kega32x_bus)
    add_test(NAME core32x_bus COMMAND core32x_bus_tests)
    add_executable(core32x_video_tests ../../tests/core32x_video_tests.c)
    target_link_libraries(core32x_video_tests PRIVATE kega32x_video)
    add_test(NAME core32x_video COMMAND core32x_video_tests)
    add_executable(sh2_idle_tests ../../tests/sh2_idle_tests.c)
    target_link_libraries(sh2_idle_tests PRIVATE kega32x_sh2)
    add_test(NAME core32x_idle COMMAND sh2_idle_tests)
    add_executable(sh2_interpreter_tests ../../tests/sh2_interpreter_tests.c)
    target_link_libraries(sh2_interpreter_tests PRIVATE kega32x_sh2)
    add_test(NAME core32x_sh2 COMMAND sh2_interpreter_tests)
    add_executable(core32x_machine_tests ../../tests/core32x_machine_tests.c)
    target_link_libraries(core32x_machine_tests PRIVATE kega32x_machine)
    add_test(NAME core32x_machine COMMAND core32x_machine_tests)
    add_executable(core32x_pwm_tests ../../tests/core32x_pwm_tests.c)
    target_link_libraries(core32x_pwm_tests PRIVATE kega32x_pwm)
    add_test(NAME core32x_pwm COMMAND core32x_pwm_tests)
    add_executable(core32x_ntsc565_tests ../../tests/core32x_ntsc565_tests.c)
    target_link_libraries(core32x_ntsc565_tests PRIVATE kega32x_ntsc565)
    add_test(NAME core32x_ntsc565 COMMAND core32x_ntsc565_tests)
    add_executable(sh2_soc_tests ../../tests/sh2_soc_tests.c)
    target_link_libraries(sh2_soc_tests PRIVATE kega32x_sh2)
    add_test(NAME core32x_soc COMMAND sh2_soc_tests)
endif()
