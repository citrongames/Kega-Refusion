cmake_minimum_required(VERSION 3.22)
project(KegaRefusion VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Multimedia)
find_package(SDL2 2.0.14 REQUIRED CONFIG)
add_library(input_backend STATIC src/input/InputManager.h src/input/InputManager.cpp)
target_include_directories(input_backend PUBLIC src)
target_compile_definitions(input_backend PUBLIC SDL_MAIN_HANDLED)
target_link_libraries(input_backend PUBLIC Qt6::Core SDL2::SDL2)
include(cmake/ArchiveDependencies.cmake)
add_library(rom_archive STATIC src/core/RomArchive.cpp src/core/RomArchive.h)
target_include_directories(rom_archive PUBLIC src)
target_link_libraries(rom_archive PUBLIC Qt6::Core PRIVATE kega_archive)
if(MSVC)
    target_compile_options(rom_archive PRIVATE /W4 /utf-8)
else()
    target_compile_options(rom_archive PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(core_host STATIC src/core/LibretroHost.cpp src/core/LibretroHost.h
    src/core/RewindBuffer.cpp src/core/RewindBuffer.h)
target_include_directories(core_host PUBLIC src third_party/libretro)
target_link_libraries(core_host PUBLIC Qt6::Core Qt6::Gui rom_archive)
add_executable(KegaRefusion WIN32
    assets/app.qrc
    src/ui/ControlsDialog.cpp src/ui/ControlsDialog.h
    src/ui/ControllerDiagram.cpp src/ui/ControllerDiagram.h
    src/main.cpp src/core/EmulationSession.cpp src/core/EmulationSession.h
    src/ui/MainWindow.cpp src/ui/MainWindow.h
    src/ui/GameView.cpp src/ui/GameView.h
    src/ui/AudioOutput.cpp src/ui/AudioOutput.h)
target_link_libraries(KegaRefusion PRIVATE core_host input_backend Qt6::Widgets Qt6::Multimedia)
if(MSVC)
    target_compile_options(core_host PRIVATE /W4 /utf-8)
    target_compile_options(KegaRefusion PRIVATE /W4 /utf-8)
else()
    target_compile_options(core_host PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(KegaRefusion PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(archive_tests tests/archive_tests.cpp)
    target_link_libraries(archive_tests PRIVATE rom_archive kega_archive)
    add_test(NAME archive_tests COMMAND archive_tests)
    add_executable(input_tests tests/input_tests.cpp)
    target_link_libraries(input_tests PRIVATE input_backend)
    add_test(NAME input_tests COMMAND input_tests)
    add_executable(osd_tests tests/osd_tests.cpp
        src/ui/GameView.cpp src/ui/GameView.h assets/app.qrc)
    target_include_directories(osd_tests PRIVATE src)
    target_link_libraries(osd_tests PRIVATE Qt6::Widgets)
    if(MSVC)
        target_compile_options(osd_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME osd_tests COMMAND osd_tests)
    if(UNIX AND NOT APPLE)
        set_tests_properties(osd_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
    add_executable(core_integration_tests tests/core_integration_tests.cpp)
    target_link_libraries(core_integration_tests PRIVATE core_host kega_archive)
    set(KEGA_TEST_CORE "" CACHE FILEPATH "Genesis Plus GX libretro binary for integration tests")
    if(KEGA_TEST_CORE)
        add_test(NAME core_integration COMMAND core_integration_tests "${KEGA_TEST_CORE}")
    endif()
endif()
install(TARGETS KegaRefusion RUNTIME DESTINATION .)

if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
    enable_language(RC)
    configure_file(cmake/windows-icon.rc.in ${CMAKE_BINARY_DIR}/windows-icon.rc @ONLY)
    set_source_files_properties(${CMAKE_BINARY_DIR}/windows-icon.rc PROPERTIES
        OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
    target_sources(KegaRefusion PRIVATE ${CMAKE_BINARY_DIR}/windows-icon.rc)
endif()
