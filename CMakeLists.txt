cmake_minimum_required(VERSION 3.22)
project(KegaRefusion VERSION 0.4.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Multimedia OpenGL OpenGLWidgets)
find_package(SDL2 2.0.14 REQUIRED CONFIG)
add_library(input_backend STATIC src/input/InputManager.h src/input/InputManager.cpp)
target_include_directories(input_backend PUBLIC src)
target_compile_definitions(input_backend PUBLIC SDL_MAIN_HANDLED)
target_link_libraries(input_backend PUBLIC Qt6::Core SDL2::SDL2)
include(cmake/ArchiveDependencies.cmake)
add_library(rom_archive STATIC src/core/RomArchive.cpp src/core/RomArchive.h)
target_include_directories(rom_archive PUBLIC src)
target_link_libraries(rom_archive PUBLIC Qt6::Core PRIVATE kega_archive)
if(MSVC)
    target_compile_options(rom_archive PRIVATE /W4 /utf-8)
else()
    target_compile_options(rom_archive PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_library(core_host STATIC src/core/LibretroHost.cpp src/core/LibretroHost.h
    src/core/RewindBuffer.cpp src/core/RewindBuffer.h
    src/core/FramePacer.cpp src/core/FramePacer.h)
target_include_directories(core_host PUBLIC src third_party/libretro)
target_link_libraries(core_host PUBLIC Qt6::Core Qt6::Gui rom_archive)
add_library(video_renderer STATIC
    src/ui/GameView.cpp src/ui/GameView.h src/ui/ShaderPipeline.cpp src/ui/ShaderPipeline.h)
target_include_directories(video_renderer PUBLIC src)
target_link_libraries(video_renderer PUBLIC Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets)
add_executable(KegaRefusion WIN32
    assets/app.qrc
    src/ui/ControlsDialog.cpp src/ui/ControlsDialog.h
    src/ui/ControllerDiagram.cpp src/ui/ControllerDiagram.h
    src/main.cpp src/core/EmulationSession.cpp src/core/EmulationSession.h
    src/ui/MainWindow.cpp src/ui/MainWindow.h
    src/ui/AudioOutput.cpp src/ui/AudioOutput.h)
target_link_libraries(KegaRefusion PRIVATE core_host input_backend video_renderer Qt6::Widgets Qt6::Multimedia)
if(MSVC)
    target_compile_options(core_host PRIVATE /W4 /utf-8)
    target_compile_options(video_renderer PRIVATE /W4 /utf-8)
    target_compile_options(KegaRefusion PRIVATE /W4 /utf-8)
else()
    target_compile_options(core_host PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(video_renderer PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(KegaRefusion PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(archive_tests tests/archive_tests.cpp)
    target_link_libraries(archive_tests PRIVATE rom_archive kega_archive)
    add_test(NAME archive_tests COMMAND archive_tests)
    add_executable(input_tests tests/input_tests.cpp)
    target_link_libraries(input_tests PRIVATE input_backend)
    add_test(NAME input_tests COMMAND input_tests)
    add_executable(osd_tests tests/osd_tests.cpp assets/app.qrc)
    target_include_directories(osd_tests PRIVATE src)
    target_link_libraries(osd_tests PRIVATE video_renderer)
    if(MSVC)
        target_compile_options(osd_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME osd_tests COMMAND osd_tests)
    if(UNIX AND NOT APPLE)
        set_tests_properties(osd_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
    add_executable(shader_tests tests/shader_tests.cpp assets/app.qrc)
    target_link_libraries(shader_tests PRIVATE video_renderer)
    option(KEGA_TEST_OPENGL "Register native GPU shader tests (requires a working display/OpenGL)" OFF)
    if(KEGA_TEST_OPENGL)
        add_test(NAME shader_gpu_tests COMMAND shader_tests)
    endif()
    add_test(NAME shader_preset_tests COMMAND shader_tests --parser-only)
    if(UNIX AND NOT APPLE)
        set_tests_properties(shader_preset_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
    add_executable(audio_queue_tests tests/audio_queue_tests.cpp)
    target_include_directories(audio_queue_tests PRIVATE src)
    target_link_libraries(audio_queue_tests PRIVATE Qt6::Core)
    add_test(NAME audio_queue_tests COMMAND audio_queue_tests)
    add_executable(audio_output_tests tests/audio_output_tests.cpp src/ui/AudioOutput.cpp src/ui/AudioOutput.h)
    target_include_directories(audio_output_tests PRIVATE src)
    target_link_libraries(audio_output_tests PRIVATE Qt6::Core Qt6::Multimedia)
    add_test(NAME audio_output_tests COMMAND audio_output_tests)
    add_executable(rewind_tests tests/rewind_tests.cpp)
    target_link_libraries(rewind_tests PRIVATE core_host)
    add_test(NAME rewind_tests COMMAND rewind_tests)
    add_executable(pacing_tests tests/pacing_tests.cpp src/core/FramePacer.cpp)
    target_include_directories(pacing_tests PRIVATE src)
    target_link_libraries(pacing_tests PRIVATE Qt6::Core)
    add_test(NAME pacing_tests COMMAND pacing_tests)
    add_executable(speed_benchmark tools/speed-benchmark.cpp)
    target_link_libraries(speed_benchmark PRIVATE core_host)
    add_executable(rewind_benchmark tools/rewind-benchmark.cpp)
    target_link_libraries(rewind_benchmark PRIVATE core_host)
    add_executable(core_integration_tests tests/core_integration_tests.cpp)
    target_link_libraries(core_integration_tests PRIVATE core_host kega_archive)
    add_executable(session_tests tests/session_tests.cpp src/core/EmulationSession.cpp src/core/EmulationSession.h)
    target_link_libraries(session_tests PRIVATE core_host Qt6::Core)
    set(KEGA_TEST_CORE "" CACHE FILEPATH "Genesis Plus GX libretro binary for integration tests")
    if(KEGA_TEST_CORE)
        add_test(NAME core_integration COMMAND core_integration_tests "${KEGA_TEST_CORE}")
        add_test(NAME session_tests COMMAND session_tests "${KEGA_TEST_CORE}")
    endif()
endif()
install(TARGETS KegaRefusion RUNTIME DESTINATION .)

if(WIN32 AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
    enable_language(RC)
    configure_file(cmake/windows-icon.rc.in ${CMAKE_BINARY_DIR}/windows-icon.rc @ONLY)
    set_source_files_properties(${CMAKE_BINARY_DIR}/windows-icon.rc PROPERTIES
        OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/app.ico")
    target_sources(KegaRefusion PRIVATE ${CMAKE_BINARY_DIR}/windows-icon.rc)
endif()
